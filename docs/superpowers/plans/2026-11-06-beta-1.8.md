/// 1.8: a click on the tab with the desktop, the taskbar or a window in front opens the panel and
/// it stays open (2026-10-14: from the desktop it closed by itself 0.4 s later). Moves the mouse.
fn tabfocus_test() {
    use windows::Win32::UI::WindowsAndMessaging::{FindWindowW, SetForegroundWindow};
    let front = |class: &str| unsafe {
        let h = FindWindowW(&windows::core::HSTRING::from(class), None).unwrap_or_default();
        let _ = SetForegroundWindow(h);
        h.0 as isize
    };
    let app = on(|c| c.enabled_apps()[0]);
    for (name, class) in [("desktop", "Progman"), ("taskbar", "Shell_TrayWnd")] {
        let mut stayed = 0;
        let mut lines = Vec::new();
        for _ in 0..10 {
            close_panel();
            front(class);
            wait(400);
            on(|c| c.show_tab(None));
            wait(1200); // past the too-early guard
            let r = on(|c| c.tab_bounds());
            let ys = on(move |c| c.edge_app_y(app)); // centre of the app's icon on the tab
            click_at(r.x + r.w / 2, ys);
            wait(1500);
            let (state, fg) = on(|c| (c.panel_state.as_str(), win32::class_name(win32::foreground_window())));
            if state == "open" {
                stayed += 1;
            }
            lines.push(format!("{state}/{fg}"));
        }
        log!("tab click from the {name}: open after 1.5 s {stayed}/10 ({}) (expect 10/10)", lines.join(" "));
    }
    close_panel();

```rust
#[derive(Debug, PartialEq, Clone, Copy)]
pub enum Sent { Confirmed, Typing, NoComposer, NotConfirmed, Busy }

pub fn can_reply(app: &str) -> bool {
    matches!(app, "instagram" | "facebook" | "discord")
}

/// Types the reply into the open conversation's composer the way a paste does (the site's own
/// input handlers run), presses Enter, and waits up to 4 s for the message to show as the last one.
const SEND_JS: &str = r#"(async (text) => {
  const boxes = () => [...document.querySelectorAll('[contenteditable="true"][role="textbox"], div[contenteditable="true"], textarea')]
    .filter((b) => { const r = b.getBoundingClientRect(); return r.width > 40 && r.height > 10; });
  let box = null;
  for (let i = 0; i < 40 && !box; i++) { box = boxes().pop() || null; if (!box) await new Promise((r) => setTimeout(r, 100)); }
  if (!box) return 'no composer';
  if ((box.value || box.textContent || '').trim()) return 'typing';
  box.focus();
  document.execCommand('insertText', false, text);
  const want = text.trim().replace(/\s+/g, ' ');
  const shown = () => [...document.querySelectorAll('[role="row"], [role="listitem"], div[dir="auto"], span[dir="auto"]')]
    .slice(-40).some((e) => (e.textContent || '').trim().replace(/\s+/g, ' ') === want && !box.contains(e));
  const enter = (type) => box.dispatchEvent(new KeyboardEvent(type, { key: 'Enter', code: 'Enter', keyCode: 13, which: 13, bubbles: true, cancelable: true }));
  enter('keydown'); enter('keypress'); enter('keyup');
  for (let i = 0; i < 40; i++) { if (shown() && !(box.value || box.textContent || '').trim()) return 'confirmed'; await new Promise((r) => setTimeout(r, 100)); }
  return 'not confirmed';
})(%TEXT%)"#;

fn send_js(text: &str) -> String {
    SEND_JS.replace("%TEXT%", &serde_json::to_string(text).unwrap_or_else(|_| "\"\"".into()))
}

fn sent_of(result: &str) -> Sent {
    match serde_json::from_str::<String>(result).as_deref() {
        Ok("confirmed") => Sent::Confirmed,
        Ok("typing") => Sent::Typing,
        Ok("no composer") => Sent::NoComposer,
        _ => Sent::NotConfirmed,
    }
}

fn sent_log(s: &Sent, text: &str) -> String {
    let what = match s { Sent::Confirmed => "confirmed", Sent::Typing => "refused: typing", Sent::NoComposer => "no composer", Sent::NotConfirmed => "not confirmed", Sent::Busy => "busy" };
    format!("sent: {} chars, {what}", text.chars().count())
}
```
